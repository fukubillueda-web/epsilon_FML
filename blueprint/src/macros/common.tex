% Manuscript source labels are external to the blueprint.
% Inside \SourceText, render \ref{...}/\eqref{...} as stable manuscript labels
% instead of trying to resolve them against the blueprint's .aux file.
\let\BlueprintOriginalSourceText\SourceText
\renewcommand{\SourceText}[1]{%
  \begingroup
  \def\ref##1{##1}%
  \def\eqref##1{##1}%
  \BlueprintOriginalSourceText{#1}%
  \endgroup
}
